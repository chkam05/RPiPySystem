ProcessInfo:
{
    "process_id": Optional[int] (pid - Process ID (unique identifier in the system)),
    "parent_process_id": Optional[int] (ppid - Parent process ID),
    "process_group_id": Optional[int] (pgid - Process group ID (group of processes with a common leader (e.g. pipelines in the shell))),
    "user_name": Optional[str] (user - Effective user of the process (who is the owner after permission changes)),
    "user_id": Optional[int] (uid - Effective user ID),
    "real_user_name": Optional[str] (ruser - Current user (the one who started the process)),
    "real_user_id": Optional[int] (ruid - Current user ID),
    "process_name": Optional[str] (comm - The name of the process (usually the basename of the program, with no arguments),
    "command_line": Optional[str] (args/cmd - The full command line of the given process, including arguments),
    "cpu_usage_percent": Optional[float] (pcpu - Percentage of CPU usage calculated by ps),
    "memory_usage_percent": Optional[float] (pmem - Percentage of RAM usage (based on RSS vs. available memory)),
    "cpu_process_time": Optional[timedelta] (cputime - Total CPU usage time by the process, (user+system)),
    "elapsed_since_start": Optional[timedelta] (etime - Time since process start (from creation to now)),
    "started_at": Optional[datetime] (start/start_time - Shortened start-up time (different format depending on how long the process runs)),
    "status": Optional[str] (stat (full) - Process status + additional flags),
    "terminal": Optional[str] (tty - Terminal assigned to the process),
    "priority": Optional[int] (pri - Kernel process priority),
    "nice_value": Optional[int] (nice - Nice value adjustable by user),
    "scheduler_class": Optional[str] (cls - Scheduler class),
    "scheduler_policy": Optional[str] (policy - Policy scheduling, scheduler type),
    "realtime_priority": Optional[str] (rtprio - Real-Time priority (when scheduler is FIFO/RR)),
    "virtual_memory_kb": Optional[int] (vsz - Process virtual memory size (KB)),
    "resident_memory_kb": Optional[int] (rss - Resident Set Size (actual memory in RAM (KB))),
    "current_cpu": Optional[int] (psr - The CPU number on which the process is currently executing),
    "cgroup_path": Optional[str] (cgroup - Cgroup assigned to the process (control group path in the system)),
    "threads": Optional[int] (nlwp - Number of LightWeight Processes (number of threads)),
    "wait_channel": Optional[str] (wchan - Symbolic name of the kernel function where the process waits ("wait channel")),
    "kernel_flags": Optional[str] (flags/f - Kernel flags of the process (in hexadecimal or decimal format)),
    "major_page_faults": Optional[int] (maj_flt - Major page faults (number of “large” page faults that required I/O)),
    "minor_page_faults": Optional[int] (min_flt - Minor page faults (“small” page faults that did not require I/O)),
    "session_id": Optional[int] (sid - Session ID - session leader (i.e. terminal session ID)),
    "thread_group_id": Optional[int] (tgid - Thread Group ID (same as the PID of the main process in a multithreaded application))
}
